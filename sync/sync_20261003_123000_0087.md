async function calDeleteEvent() {
    if (!calDetailEvent) return;

### ✅ Odwołaj (`calCancelEvent()`) — DZIAŁA, pełna implementacja

```js
async function calCancelEvent() {
    if (!calDetailEvent) return;
    if (!confirm('Odwołać to wydarzenie? Wszyscy zaproszeni dostaną powiadomienie.')) return;
    const ev = calDetailEvent;
    const evId = ev.eventId || ev.id;
    try {
        await db.collection('events').doc(evId).update({ status: 'CANCELLED', cancelledAt: new Date().toISOString(), cancelledBy: getCurrentUserId() });
        const idx = calAllEvents.findIndex(e => (e.eventId||e.id) === evId);
        if (idx >= 0) calAllEvents[idx] = { ...calAllEvents[idx], status: 'CANCELLED' };
        openCalDetailPanel({ ...ev, status: 'CANCELLED' });
        renderCalGrid();
        const d = calSelectedDate;
        if (d) renderDayInspector(d);
        if (typeof clearTeamEventsCache === 'function') clearTeamEventsCache(_activeTeamId);
        await refreshDashboard();
    } catch(e) { console.error('❌ calCancelEvent', e); alert('Błąd: ' + e.message); }
}
```
`status: 'CANCELLED'`, powiadomienie `EVENT_CANCELLED` wysyła Cloud Function
`onEventUpdated` (nie klient) do wszystkich zaproszonych.
