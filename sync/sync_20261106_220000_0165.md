## 1. Model danych (to, co leży w Firestore)

Dokument `events/{eventId}`, pole `matchData`:

```
matchData.matchStatus   : 'UPCOMING' | 'LIVE' | 'FINISHED'
matchData.clockSegments : [{ start: ISOString, end: ISOString | null }, ...]
matchData.endedAt       : ISOString | null
matchData.result        : { our, opponent, outcome, updatedBy, updatedAt }
updatedAt               : serverTimestamp()   // przy każdej akcji
```

Zasada: **jeden odcinek gry = jedno `{start, end}`**. Start tworzy odcinek otwarty (`end: null`).
Przerwa zamyka ostatni odcinek. Wznowienie dopisuje nowy otwarty. Koniec zamyka otwarty odcinek
(`end === endedAt`). Cofnięcie zakończenia: jeśli `last.end === endedAt` → otwiera ten sam odcinek,
inaczej dopisuje nowy. **Czas to wyłącznie suma `(end || teraz) - start` po odcinkach — w Firestore
nigdy nie zapisujemy "ile minęło".** Dlatego zegar nie wymaga żadnych zapisów podczas gry.

## 2. Akcje (zapisy) — jedyne momenty, kiedy licznik dotyka Firestore

Wersja `compat` (db = `firebase.firestore()`, `TS = firebase.firestore.FieldValue.serverTimestamp()`):

```js
const evRef = (id) => db.collection('events').doc(id);

// START
function startMatch(eventId) {
  return evRef(eventId).update({
    'matchData.matchStatus': 'LIVE',
    'matchData.clockSegments': [{ start: new Date().toISOString(), end: null }],
    updatedAt: TS,
  });
}

// PRZERWA — zamyka bieżący odcinek
async function pauseMatch(eventId, segs /* z ostatniego snapshotu, patrz uwaga niżej */) {
  segs = segs.map(x => ({ ...x }));
  if (!segs.length || segs[segs.length - 1].end) return;
  segs[segs.length - 1].end = new Date().toISOString();
  return evRef(eventId).update({ 'matchData.clockSegments': segs, updatedAt: TS });
}

// WZNÓW — nowy odcinek
async function resumeMatch(eventId, segs) {
  segs = segs.map(x => ({ ...x }));
  if (segs.length && !segs[segs.length - 1].end) return;
  segs.push({ start: new Date().toISOString(), end: null });
  return evRef(eventId).update({ 'matchData.clockSegments': segs, updatedAt: TS });
}

// KONIEC — zamyka otwarty odcinek, ustawia wynik
async function endMatch(eventId, segs, our, opponent, userId) {
  const outcome = our > opponent ? 'WIN' : our < opponent ? 'LOSS' : 'DRAW';
  const nowIso = new Date().toISOString();
  segs = segs.map(x => ({ ...x }));
  if (segs.length && !segs[segs.length - 1].end) segs[segs.length - 1].end = nowIso;
  return evRef(eventId).update({
    ...(segs.length ? { 'matchData.clockSegments': segs } : {}),
    'matchData.matchStatus': 'FINISHED',
    'matchData.endedAt': nowIso,
    'matchData.endedBy': userId,
    'matchData.result': { our, opponent, outcome, updatedBy: userId, updatedAt: nowIso },
    updatedAt: TS,
  });
}

// COFNIJ ZAKOŃCZENIE
async function reopenMatch(eventId, md /* matchData z ostatniego snapshotu */) {
  const segs = (md.clockSegments || []).map(x => ({ ...x }));
  if (!segs.length) segs.push({ start: new Date().toISOString(), end: null });
  else {
    const last = segs[segs.length - 1];
    if (last.end && md.endedAt && last.end === md.endedAt) last.end = null;
    else segs.push({ start: new Date().toISOString(), end: null });
  }
  return evRef(eventId).update({
    'matchData.matchStatus': 'LIVE', 'matchData.endedAt': null, 'matchData.endedBy': null,
    'matchData.clockSegments': segs, updatedAt: TS,
  });
}
```

**Uwaga (uczciwie, nasz drobny koszt):** w appce `pauseMatch`/`resumeMatch`/`endMatch` robią przed
zapisem dodatkowy `get()` dokumentu, żeby wczytać aktualne odcinki (1 odczyt na akcję). W wersji
powyżej przekazujemy `segs` z ostatniego snapshotu listenera — **zero dodatkowych odczytów**. Appkę
też o to poprawimy, jeśli test pokaże, że ma to znaczenie.

## 3. Wyświetlanie licznika — tykanie w 100% lokalne

To jest cała logika zegara. **Nie ma tu żadnego wywołania Firestore.**

```js
const MAX_MIN = 240; // limit 4 h łącznego czasu gry -> "240:00+"

function fmt(ms) {
  const capped = ms >= MAX_MIN * 60000;
  const total = Math.floor(Math.min(Math.max(0, ms), MAX_MIN * 60000) / 1000);
  const mm = Math.floor(total / 60), ss = total % 60;
  return String(mm).padStart(2, '0') + ':' + String(ss).padStart(2, '0') + (capped ? '+' : '');
}

function segMs(sg, now) {
  const a = new Date(sg.start).getTime();
  const b = sg.end ? new Date(sg.end).getTime() : now;
  return (isNaN(a) || isNaN(b)) ? 0 : Math.max(0, b - a);
}

// segs: matchData.clockSegments, live: matchStatus === 'LIVE'
function computeClock(segs, live, now) {
  segs = segs || [];
  const running = live && segs.length > 0 && !segs[segs.length - 1].end;
  const paused  = live && !running;
  const total   = segs.reduce((acc, sg) => acc + segMs(sg, now), 0);        // suma cykli (przerwy się nie liczą)
  const cycle   = segs.length ? segMs(segs[segs.length - 1], now) : 0;      // bieżący (lub ostatni) cykl
  return { running, paused, total, cycle };
}

// RENDER: setInterval co 1 s, TYLKO podmienia tekst w DOM. Zero Firestore.
let clockTimer = null;
function startClockTicker(getMatchData, el) {
  stopClockTicker();
  const tick = () => {
    const md = getMatchData();                      // obiekt trzymany w pamięci (ostatni snapshot)
    const segs = md.clockSegments || [];
    const live = md.matchStatus === 'LIVE';
    if (!segs.length) { el.textContent = ''; return; }
    const c = computeClock(segs, live, Date.now());
    el.textContent = (c.paused ? '⏸ ' : '⏱ ') + fmt(c.cycle);   // pill = bieżący cykl
  };
  tick();
  if (getMatchData().matchStatus === 'LIVE') clockTimer = setInterval(tick, 1000);
}
function stopClockTicker() { if (clockTimer) { clearInterval(clockTimer); clockTimer = null; } }
```

W appce ticker chodzi tylko gdy `matchStatus === 'LIVE'` i są odcinki; przy FINISHED/UPCOMING nie tyka.
Dodatkowo (ekran Mecz) pokazujemy "cały mecz" = `fmt(total)` oraz zwijaną listę czasów cykli
(`fmt(segMs(sg, now))` dla zamkniętych odcinków).

## 4. Status LIVE — jeden listener, nie polling

```js
let unsubLive = null;
let lastMatchData = {};                 // trzymane w pamięci — z niego liczy ticker i akcje

function attachLiveListener(eventId, onChange) {
  detachLiveListener();
  let first = true;
  unsubLive = db.collection('events').doc(eventId).onSnapshot((doc) => {
    if (!doc.exists) return;
    lastMatchData = doc.data().matchData || {};
    if (first) { first = false; }        // pierwszy snapshot = to, co już wczytaliśmy; odczyt i tak jest policzony
    onChange(lastMatchData);             // przerysuj kartę (wynik, status) — ticker sam bierze lastMatchData
  }, (err) => { if (err.code !== 'permission-denied') console.error(err); });
}
function detachLiveListener() { if (unsubLive) { unsubLive(); unsubLive = null; } }
```

Podpinamy **tylko** gdy mecz jest widoczny i `matchStatus !== 'FINISHED'`; odpinamy przy zmianie
meczu, FINISHED i opuszczeniu ekranu. Koszt: **1 odczyt przy podpięciu + 1 odczyt na każdą zmianę
dokumentu** (start/przerwa/wznów/gol/koniec), nie zależy od czasu trwania meczu.

## 5. Protokół testu odczytów (propozycja — do uzgodnienia)

Cel: sprawdzić, czy sam licznik generuje odczyty rosnące z czasem. Oczekiwanie (z kodu): **nie**.

**Warunki:** 1 użytkownik (trener), 1 mecz testowy dzisiejszy, jedna karta/ekran z meczem otwarta przez
cały test, bez nawigacji i bez odświeżania. Włączony `fsDebugLog` (WEB) — u nas dopiszemy równoważny
licznik odczytów/zapisów (ten sam format logu), żeby porównywać jednakowe liczby, a nie wykres z
konsoli Firebase (ma opóźnienie i miesza wszystko, co użytkownik robił).

**Scenariusz (każda platforma osobno):**
1. Wejść na ekran meczu. Zapisać licznik odczytów = **R0**.
2. Start meczu → **R1** (oczekiwane: +1 odczyt z `onSnapshot` na zmianę, ewentualnie +1 jeśli w kodzie jest `get()` przed zapisem).
3. **Odczekać 10 minut** z włączonym tykającym licznikiem, bez żadnej akcji → **R2**.
4. Przerwa → Wznów → Koniec → **R3**.

**Kryterium:** `R2 - R1` powinno wynosić **0** (licznik przez 10 minut nie robi żadnych odczytów).
Cała reszta to odczyty zdarzeń, rzędu pojedynczych sztuk. Jeśli `R2 - R1 > 0` — znaleźliśmy nieznane
źródło odczytów (po naszej lub Waszej stronie) i wtedy dopiero się nad tym pochylamy.

**Dodatkowo (żeby porównać z "podejrzanym" 3 listopada):** to samo z drugim urządzeniem jako widzem
(rodzic, ekran Start otwarty z kartą meczu LIVE) — oczekiwane: +1 odczyt na każdą akcję trenera.

Prosimy o potwierdzenie, że WEB wdroży powyższy kod 1:1 (najlepiej na osobnej gałęzi/flagi), i o wyniki
z `fsDebugLog`. My równolegle dopiszemy licznik odczytów w buildzie testowym appki.
